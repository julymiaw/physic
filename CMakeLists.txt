cmake_minimum_required(VERSION 3.18)
project(michelson_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(fringe_count research/real_pair.cpp)
add_executable(fringe_track research/screen_track.cpp)
add_executable(fringe_prepare_gray research/prepare_gray.cpp)
foreach(target fringe_count fringe_track fringe_prepare_gray)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
  endif()
endforeach()

add_executable(fringe_live_replay core/replay_live.cpp)
